Add Daily Reward

checkDailyReward() {
    const lastClaim = localStorage.getItem('lastDailyReward');
    const today = new Date().toDateString();

    if (lastClaim !== today) {
        const reward = 1000;
        this.balance += reward;
        localStorage.setItem('lastDailyReward', today);

        tg.showPopup({
            title: '🎁 Daily Reward',
            message: `You received ${reward} 💎!`,
            buttons: [{type: 'ok'}]
        });
    }
}

// Call in init()
this.checkDailyReward();

Add Combo System

// Add to constructor
this.combo = 0;
this.comboTimer = null;

// In handleTap()
this.combo++;
clearTimeout(this.comboTimer);

// Bonus for combo
const comboBonus = Math.floor(this.combo / 10);
this.balance += this.tapPower + comboBonus;

// Reset combo after 2 seconds
this.comboTimer = setTimeout(() => {
    this.combo = 0;
}, 2000);

Share Score

shareScore() {
    const url = `https://t.me/share/url?url=${encodeURIComponent(
        'https://t.me/YourBot/clicker'
    )}&text=${encodeURIComponent(

Referral System

// Get referrer from URL
const urlParams = new URLSearchParams(window.location.search);
const referrerId = urlParams.get('ref');

if (referrerId && !localStorage.getItem('referrer')) {
    localStorage.setItem('referrer', referrerId);

    // Give bonus to new user
    this.balance += 500;

    // Send reward to referrer (requires backend)
    fetch('/api/referral', {
        method: 'POST',
        body: JSON.stringify({
            referrer: referrerId,
            newUser: tg.initDataUnsafe.user.id
        })
    });
}

📊 Analytics Integration

Track Events

trackEvent(eventName, data = {}) {
    // Google Analytics
    if (window.gtag) {
        gtag('event', eventName, data);
    }

    // Telegram Analytics
    if (tg.sendData) {
        tg.sendData(JSON.stringify({
            event: eventName,
            ...data,
            userId: tg.initDataUnsafe.user?.id
        }));
    }
}

// Usage
this.trackEvent('tap', { balance: this.balance });
this.trackEvent('upgrade_purchased', { type: 'energy', level: this.boosts.energy.level });

🎮 Game Mechanics

Add Power-ups

activatePowerUp(type) {
    const powerUps = {
        double: { duration: 30000, multiplier: 2 },
        triple: { duration: 15000, multiplier: 3 },
        mega: { duration: 10000, multiplier: 5 }
    };

    const powerUp = powerUps[type];
    const originalPower = this.tapPower;

    this.tapPower *= powerUp.multiplier;

    setTimeout(() => {
        this.tapPower = originalPower;
        tg.showAlert('Power-up expired!');
    }, powerUp.duration);
}
